        async function loadConfig() {
            const loadingOverlay = document.getElementById('loadingOverlay');
            const errorMessage = document.getElementById('errorMessage');
    
            try {
                loadingOverlay.style.display = 'flex';
                const response = await fetch('./Config.json');
        
                if (!response.ok) throw new Error(`HTTP ${response.status}: ConfigFile loading Failed`);
                config = await response.json();
                if (!config.testHierarchy) throw new Error('No testHierarchy in config file');
        
                console.log('✅ ConfigFile Loaded successfully');
                initializePage();
                loadingOverlay.style.display = 'none';
        
            } catch (error) {

                errorMessage.style.display = 'block';

            }
        }

	   /**
	 * 验证参数值格式
	 * @param {string} limitType - Limit类型（Linear, Const, Log, Single, Poly）
	 * @param {string} value - 输入的值
	 * @returns {boolean} - 是否有效
	 */
	function validateParameterFormat(limitType, value) {
	    if (!value || !value.trim()) {
	        return true; // 空值允许（可选字段）
	    }
	    
	    value = value.trim();
	    
	    switch(limitType) {
	        case 'Const':
	        case 'Single':
	            // 只能是单个数字（可以是负数、小数）
	            return /^-?\d+(\.\d+)?$/.test(value);
	            
	        case 'Log':
	        case 'Linear':
	        case 'Poly':
	            // 必须是两个数字，用逗号分隔
	            const parts = value.split(',');
	            if (parts.length !== 2) return false;
	            return parts.every(part => /^-?\d+(\.\d+)?$/.test(part.trim()));
	            
	        default:
	            return true;
	    }
	}
	
	/**
	 * 获取参数格式提示文本
	 */
	function getFormatHint(limitType) {
	    switch(limitType) {
	        case 'Const':
	        case 'Single':
	            return 'Enter single value (e.g., 79)';
	        case 'Log':
	        case 'Linear':
	        case 'Poly':
	            return 'Enter two values separated by comma (e.g., 79,73)';
	        default:
	            return '';
	    }
	}
	
	/**
	 * 验证输入框并添加视觉反馈
	 */
	function validateInput(input, limitType) {
	    const value = input.value.trim();
	    const isValid = validateParameterFormat(limitType, value);
	    
	    if (!isValid && value !== '') {
	        // 错误状态
	        input.style.borderColor = '#f44336';
	        input.style.boxShadow = '0 0 5px rgba(244, 67, 54, 0.5)';
	        input.style.backgroundColor = '#FFEBEE';
	        input.setAttribute('data-invalid', 'true');
	        return false;
	    } else {
	        // 正确状态
	        if (value !== '') {
	            input.style.borderColor = '#4CAF50';
	            input.style.boxShadow = '0 0 5px rgba(76, 175, 80, 0.3)';
	            input.style.backgroundColor = '#E8F5E9';
	        } else {
	            clearValidationStyle(input);
	        }
	        input.removeAttribute('data-invalid');
	        return true;
	    }
	}
	
	/**
	 * 清除验证样式
	 */
	function clearValidationStyle(input) {
	    input.style.borderColor = '';
	    input.style.boxShadow = '';
	    input.style.backgroundColor = '';
	    input.removeAttribute('data-invalid');
	}
	
	/**
	 * 为参数表格设置实时验证
	 */
	function setupParameterValidation(row) {
	    // ========== 1. Emission 测试：PK/QP/AV + LimitType ==========
	    const emissionPairs = [
	        { typeKey: 'pkLimitType', valueKey: 'pk' },
	        { typeKey: 'qpLimitType', valueKey: 'qp' },
	        { typeKey: 'avLimitType', valueKey: 'av' }
	    ];
	    
	    emissionPairs.forEach(pair => {
	        const typeSelect = row.querySelector(`select[data-key="${pair.typeKey}"]`);
	        const valueInput = row.querySelector(`input[data-key="${pair.valueKey}"]`);
	        
	        if (typeSelect && valueInput) {
	            setupValidationPair(typeSelect, valueInput);
	        }
	    });
	    
	    // ========== 2. Immunity 测试：limitCalculationType + limitValue1/2 ==========
	    // 🔥 关键修改：使用 querySelectorAll 找到所有的 limitCalculationType
	    const limitCalcSelects = row.querySelectorAll('select[data-key="limitCalculationType"]');
	    
	    limitCalcSelects.forEach(limitCalcSelect => {
	        // 找到当前 limitCalculationType 所在的单元格
	        const currentCell = limitCalcSelect.closest('td');
	        if (!currentCell) return;
	        
	        // 获取当前单元格的索引
	        const cellIndex = currentCell.cellIndex;
	        
	        // 找到后续两个单元格（应该是 limitValue1 和 limitValue2）
	        const cells = row.cells;
	        
	        // limitValue1 应该在 limitCalculationType 后面第 1 列
	        const limitValue1Cell = cells[cellIndex + 1];
	        const limitValue1 = limitValue1Cell ? limitValue1Cell.querySelector('input[data-key="limitValue1"]') : null;
	        
	        // acceptanceCriteria1 在第 2 列（跳过）
	        
	        // limitValue2 应该在 limitCalculationType 后面第 3 列
	        const limitValue2Cell = cells[cellIndex + 3];
	        const limitValue2 = limitValue2Cell ? limitValue2Cell.querySelector('input[data-key="limitValue2"]') : null;
	        
	        // 为找到的输入框设置验证
	        if (limitValue1) {
	            setupValidationPair(limitCalcSelect, limitValue1);
	        }
	        
	        if (limitValue2) {
	            setupValidationPair(limitCalcSelect, limitValue2);
	        }
	    });
	}
	
	/**
	 * 为一对 select 和 input 设置验证
	 */
	function setupValidationPair(typeSelect, valueInput) {
	    // 监听类型下拉框变化
	    typeSelect.addEventListener('change', function() {
	        const limitType = this.value;
	        valueInput.placeholder = getFormatHint(limitType);
	        
	        // 如果输入框已有值，立即验证
	        if (valueInput.value) {
	            validateInput(valueInput, limitType);
	        } else {
	            clearValidationStyle(valueInput);
	        }
	    });
	    
	    // 监听值输入框实时输入
	    valueInput.addEventListener('input', function() {
	        const limitType = typeSelect.value;
	        if (limitType) {
	            validateInput(this, limitType);
	        }
	    });
	    
	    // 监听失去焦点事件
	    valueInput.addEventListener('blur', function() {
	        const limitType = typeSelect.value;
	        if (limitType && this.value) {
	            validateInput(this, limitType);
	        }
	    });
	    
	    // 初始化：如果已有值，触发验证
	    if (typeSelect.value && valueInput.value) {
	        validateInput(valueInput, typeSelect.value);
	    }
	}
	/**
	 * 生成 Excel 导出文件名
	 * 格式：客户名称_Level2_Level3_Level4_Specification_SpecDate_YYYYMMDD_HHMMSS.xlsx
	 */
	function generateExcelFileName() {
	    // ========== 1. 获取客户名称 ==========
	    const customer = document.getElementById('customer').value.trim() || 'UnknownCustomer';
	    
	    // ========== 2. 获取测试项信息 ==========
	    const level2 = selectedLevel2 || 'NoLevel2';
	    const level3 = selectedLevel3 || '';
	    const level4 = selectedLevel4 || '';
	    
	    // ========== 3. 获取规格信息 ==========
	    const specification = document.getElementById('specification').value.trim() || 'NoSpec';
	    const specDate = document.getElementById('specification_date').value.trim() || '';
	    
	    // ========== 4. 获取当前日期时间（格式：YYYYMMDD_HHMMSS）==========
	    const now = new Date();
	    const year = now.getFullYear();
	    const month = String(now.getMonth() + 1).padStart(2, '0');
	    const day = String(now.getDate()).padStart(2, '0');
	    const hours = String(now.getHours()).padStart(2, '0');
	    const minutes = String(now.getMinutes()).padStart(2, '0');
	    const seconds = String(now.getSeconds()).padStart(2, '0');
	    const currentDateTime = `${year}${month}${day}_${hours}${minutes}${seconds}`;
	    
	    // ========== 5. 清理特殊字符（避免文件名非法）==========
	    const cleanString = (str) => {
	        return str.replace(/[<>:"/\\|?*\s]/g, '_'); // 替换Windows不允许的字符和空格
	    };
	    
	    // ========== 6. 构建文件名各部分 ==========
	    const parts = [
	        cleanString(customer),           // 客户名称
	        cleanString(level2)              // Level 2
	    ];
	    
	    if (level3) parts.push(cleanString(level3));  // Level 3（如果有）
	    if (level4) parts.push(cleanString(level4));  // Level 4（如果有）
	    
	    parts.push(cleanString(specification));  // Specification
	    
	    if (specDate) {
	        parts.push(cleanString(specDate));  // Specification Date（如果有）
	    }
	    
	    parts.push(currentDateTime);  // 当前日期时间
	    
	    // ========== 7. 拼接文件名 ==========
	    const fileName = parts.join('_') + '.xlsx';
	    
	    console.log('📄 生成文件名:', fileName);
	    return fileName;
	}

        function addCTIRow(data = {}) {
            const tbody = document.getElementById('generalTableBody');
            const row = tbody.insertRow();
            const columns = window.currentGeneralColumns || [];
            columns.forEach(col => {
                const cell = row.insertCell();
                if (col.type === 'select' && col.options) {
                    const select = document.createElement('select');
                    select.setAttribute('data-key', col.key);
                    if (isRequiredParameter(col.key)) {
                        select.required = true;
                        select.style.borderColor = '#FF9800';
                        select.style.backgroundColor = '#FFF9C4';
                    }
                    const emptyOption = document.createElement('option');
                    emptyOption.value = '';
                    emptyOption.textContent = '-- Select --';
                    select.appendChild(emptyOption);
                    col.options.forEach(option => {
                        const opt = document.createElement('option');
                        opt.value = option;
                        opt.textContent = option;
                        if (data[col.key] === option) opt.selected = true;
                        select.appendChild(opt);
                    });
                    cell.appendChild(select);
                } else {
                    const input = document.createElement('input');
                    input.type = col.type || 'text';
                    input.setAttribute('data-key', col.key);
                    input.value = data[col.key] || '';
                    input.placeholder = col.placeholder || '';
                    if (isRequiredParameter(col.key)) {
                        input.required = true;
                        input.style.borderColor = '#FF9800';
                        input.style.backgroundColor = '#FFF9C4';
                    }
                    cell.appendChild(input);
                }
            });
            const actionCell = row.insertCell();
            actionCell.style.whiteSpace = 'nowrap';
            const copyBtn = document.createElement('button');
            copyBtn.className = 'btn-copy';
            copyBtn.textContent = 'Copy';
            copyBtn.onclick = function() { copyCTIRow(this); };
            const deleteBtn = document.createElement('button');
            deleteBtn.className = 'btn-remove';
            deleteBtn.textContent = 'Delete';
            deleteBtn.onclick = function() { removeCTIRow(this); };
            actionCell.appendChild(copyBtn);
            actionCell.appendChild(deleteBtn);
        }

        function copyCTIRow(btn) {
            const currentRow = btn.closest('tr');
            const inputs = currentRow.querySelectorAll('input[data-key], select[data-key]');
            const rowData = {};
            inputs.forEach(input => rowData[input.getAttribute('data-key')] = input.value);
            addCTIRow(rowData);
            const newRow = document.getElementById('generalTableBody').lastElementChild;
            newRow.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
            newRow.style.backgroundColor = '#E8F5E9';
            setTimeout(() => { newRow.style.backgroundColor = ''; }, 1000);
            showNotification('✅ Already copy and add new row');
        }

        function removeCTIRow(btn) {
            const tbody = btn.closest('tbody');
            if (tbody.rows.length <= 1) {
                showNotification('⚠️ Minimum keep 1 row', 'error');
                return;
            }
            btn.closest('tr').remove();
            showNotification('✅ Row Deleted');
        }

	function generateLimitTable(limitConfig, columnGroupsOrColumns) {
	    const table = document.getElementById('limitTable');
	    const thead = table.querySelector('thead');
	    const tbody = document.getElementById('limitTableBody');
	    thead.innerHTML = '';
	    tbody.innerHTML = '';
	
	    let columns = [];
	
	    // ========== CTI 测试：直接传列定义数组 ==========
	    if (Array.isArray(columnGroupsOrColumns) &&
	        columnGroupsOrColumns.length > 0 &&
	        typeof columnGroupsOrColumns[0] === 'object' &&
	        columnGroupsOrColumns[0].key) {
	        columns = columnGroupsOrColumns;
	        document.getElementById('addGroupBtn').style.display = 'none';
	        document.getElementById('addLevelBtn').style.display = 'none';
	    } 
	    // ========== Immunity 测试：动态 Group + Level ==========
	    else if (limitConfig && Array.isArray(columnGroupsOrColumns)) {
	        // 1. 先加载 base 列
	        let baseColumns = [];
	        columnGroupsOrColumns.forEach(groupKey => {
	            if (groupKey === 'base' && limitConfig.base) {
	                baseColumns = limitConfig.base;
	            }
	        });
	
	        if (baseColumns.length === 0) {
	            tbody.innerHTML = '<tr><td colspan="100" style="text-align:center;color:#999;">无参数配置</td></tr>';
	            return;
	        }
	
	        // 2. 初始化 Group 和 Level 配置（默认 1 个 Group，每个 Group 2 个 Level）
	        groupLevelConfig = { 1: DEFAULT_LEVEL_COUNT };
	        
	        const groupTemplate = limitConfig.groupTemplate;
	        if (!groupTemplate) {
	            console.error('❌ 缺少 groupTemplate 配置');
	            return;
	        }
	        
	        columns = [...baseColumns];
	        
	        // 3. 生成 Group 和 Level 列
	        for (const [groupIndex, levelCount] of Object.entries(groupLevelConfig)) {
	            const groupNum = parseInt(groupIndex);
	            
	            // 添加 Limit_CalculationType（每个 Group 1 个）
	            columns.push({
	                ...groupTemplate.limitCalculationType,
	                key: `${groupTemplate.limitCalculationType.key}_group${groupNum}`,
	                label: `Group ${groupNum} - ${groupTemplate.limitCalculationType.label}`,
	                groupIndex: groupNum,
	                isGroupHeader: true
	            });
	            
	            // 添加该 Group 的所有 Level
	            for (let levelIdx = 1; levelIdx <= levelCount; levelIdx++) {
	                groupTemplate.levelTemplate.forEach(col => {
	                    columns.push({
	                        ...col,
	                        key: `${col.key}_group${groupNum}_level${levelIdx}`,
	                        label: `Group ${groupNum}_Level ${levelIdx}_${col.label}`,
	                        groupIndex: groupNum,
	                        levelIndex: levelIdx
	                    });
	                });
	            }
	        }
	
	        // 显示按钮
	        document.getElementById('addGroupBtn').style.display = 'inline-block';
	        document.getElementById('addLevelBtn').style.display = 'inline-block';
	    }
	
	    if (columns.length === 0) {
	        tbody.innerHTML = '<tr><td colspan="100" style="text-align:center;color:#999;">无参数配置</td></tr>';
	        return;
	    }
	
	    // ========== 生成表头 ==========
	    const headerRow = document.createElement('tr');
	    columns.forEach(col => {
	        const th = document.createElement('th');
	        
	        // ✅ Group Header（Limit_CalculationType）- 添加删除 Group 按钮
	        if (col.isGroupHeader) {
	            th.innerHTML = `
	                ${col.label}
	                <button class="btn-remove" onclick="removeGroupColumn(${col.groupIndex})" 
	                        style="margin-left: 5px; padding: 2px 6px; font-size: 9px;">
	                    ✕ Group
	                </button>
	            `;
	        }
	        // ✅ Level Header（第一个 Level 字段）- 添加删除 Level 按钮
	        else if (col.levelIndex && col.key.includes('limitValue')) {
	            th.innerHTML = `
	                ${col.label}
	                <button class="btn-remove" onclick="removeLevelColumn(${col.groupIndex}, ${col.levelIndex})"

	function addGroupColumn() {
	    const totalGroups = Object.keys(groupLevelConfig).length;
	
	    if (totalGroups >= MAX_GROUPS) {
	        showNotification(`⚠️ Maximum ${MAX_GROUPS} Groups reached.`, 'error');
	        return;
	    }
	
	    // 找到下一个可用的 Group 编号
	    let nextGroupIndex = 1;
	    while (groupLevelConfig[nextGroupIndex]) {
	        nextGroupIndex++;
	    }
	
	    // 增加新的 Group（默认 2 个 Level）
	    groupLevelConfig[nextGroupIndex] = DEFAULT_LEVEL_COUNT;
	
	    console.log('➕ 添加 Group:', nextGroupIndex, groupLevelConfig);
	
	    // 重新生成表格
	    regenerateTable();
	
	    showNotification(`✅ Added Group ${nextGroupIndex} with ${DEFAULT_LEVEL_COUNT} Levels.`);
	}

	function addLevelColumn() {
	    const maxLevelCount = Math.max(...Object.values(groupLevelConfig));
	    
	    if (maxLevelCount >= MAX_LEVELS_PER_GROUP) {
	        showNotification(`⚠️ Maximum ${MAX_LEVELS_PER_GROUP} Levels reached!`, 'error');
	        return;
	    }
	
	    // 为每个 Group 增加一个 Level
	    Object.keys(groupLevelConfig).forEach((groupKey) => {
	        groupLevelConfig[groupKey]++;
	    });
	
	    console.log('➕ 添加 Level 到所有 Group:', groupLevelConfig);
	
	    // 重新生成表格
	    regenerateTable();
	
	    showNotification(`✅ Added Level ${maxLevelCount + 1} to all groups.`);
	}

	function removeGroupColumn(groupIndex) {
	    const totalGroups = Object.keys(groupLevelConfig).length;
	    
	    if (totalGroups <= 1) {
	        showNotification('⚠️ At least keep 1 Group', 'error');
	        return;
	    }
	
	    if (!confirm(`🗑️ Delete Group ${groupIndex}?\n\nThis will remove Group ${groupIndex} and all its levels.`)) {
	        return;
	    }
	
	    // 直接删除指定的 Group
	    delete groupLevelConfig[groupIndex];
	
	    console.log('🗑️ 删除 Group:', groupIndex, '剩余:', groupLevelConfig);
	
	    // 重新生成表格
	    regenerateTable();
	
	    showNotification(`✅ Deleted Group ${groupIndex}`);
	}

	function removeLevelColumn(groupIndex, levelIndex) {
	    // ✅ 检查所有 Group 的最小 Level 数量
	    const minLevelCount = Math.min(...Object.values(groupLevelConfig));
	    
	    if (minLevelCount <= 1) {
	        showNotification('⚠️ At least keep 1 level in each group', 'error');
	        return;
	    }
	    
	    if (!confirm(`🗑️ Delete Level ${levelIndex} from ALL groups?\n\nThis will remove Level ${levelIndex} from all ${Object.keys(groupLevelConfig).length} group(s).`)) {
	        return;
	    }
	    
	    // ✅ 从所有 Group 中减少一个 Level
	    Object.keys(groupLevelConfig).forEach(gIndex => {
	        if (groupLevelConfig[gIndex] > 0) {
	            groupLevelConfig[gIndex]--;
	        }
	    });
	    
	    console.log('🗑️ 从所有 Group 删除 Level:', levelIndex, '剩余:', groupLevelConfig);
	    
	    // 重新生成表格
	    regenerateTable();
	    
	    showNotification(`✅ Deleted Level ${levelIndex} from all groups`);
	}

        function addGeneralRowDynamic(data = {}) {
            const tbody = document.getElementById('generalTableBody');
            const row = tbody.insertRow();
            const columns = window.currentGeneralColumns || [];
            columns.forEach(col => {
                const cell = row.insertCell();
                if (col.type === 'select' && col.options) {
                    const select = document.createElement('select');
                    select.setAttribute('data-key', col.key);
                    const emptyOption = document.createElement('option');
                    emptyOption.value = '';
                    emptyOption.textContent = col.placeholder || 'Pls select';
                    select.appendChild(emptyOption);
                    col.options.forEach(option => {
                        const opt = document.createElement('option');
                        opt.value = option;
                        opt.textContent = option;
                        if (data[col.key] === option) opt.selected = true;
                        select.appendChild(opt);
                    });
                    if (data[col.key] && !col.options.includes(data[col.key])) select.value = '';
                    cell.appendChild(select);
                } else {
                    const input = document.createElement('input');
                    input.type = 'text';
                    input.setAttribute('data-key', col.key);
                    input.value = data[col.key] || '';
                    input.placeholder = col.placeholder || '';
                    cell.appendChild(input);
                }
            });
            const actionCell = row.insertCell();
            actionCell.style.whiteSpace = 'nowrap';
            const copyBtn = document.createElement('button');
            copyBtn.className = 'btn-copy';
            copyBtn.textContent = 'Copy';
            copyBtn.onclick = function() { copyRow(this, 'general'); };
            const deleteBtn = document.createElement('button');
            deleteBtn.className = 'btn-remove';
            deleteBtn.textContent = 'Delete';
            deleteBtn.onclick = function() { removeRow(this); };
            actionCell.appendChild(copyBtn);
            actionCell.appendChild(deleteBtn);
	    
	    setupParameterValidation(row);
		
        }

		function addLimitRowDynamic(data = {}) {
			  const tbody = document.getElementById('limitTableBody');
			  const row = tbody.insertRow();
			  const columns = window.currentLimitColumns || [];
			  if (columns.length === 0) return;

			  columns.forEach(col => {
				const cell = row.insertCell();

				if (col.type === 'select' && col.options) {
				  const select = document.createElement('select');
				  select.className = 'param-input';
				  select.setAttribute('data-key', col.key);

				  // CTI 必填高亮（依赖 waveformParameterMapping.requiredParams）
				  if (selectedLevel1 === 'Transient_Coupled_Immunity' && isRequiredParameter(col.key)) {
					select.required = true;
				  }

				  const emptyOption = document.createElement('option');
				  emptyOption.value = '';
				  emptyOption.textContent = col.placeholder || 'Pls Select';
				  select.appendChild(emptyOption);

				  col.options.forEach(option => {
					const opt = document.createElement('option');
					opt.value = option;
					opt.textContent = option;
					if (data[col.key] === option) opt.selected = true;
					select.appendChild(opt);
				  });

				  cell.appendChild(select);
				} else {
				  const input = document.createElement('input');
				  input.className = 'param-input';
				  input.type = col.type || 'text';
				  input.setAttribute('data-key', col.key);
				  input.value = data[col.key] || '';
				  input.placeholder = col.placeholder || '';

				  // CTI 必填高亮（依赖 waveformParameterMapping.requiredParams）
				  if (selectedLevel1 === 'Transient_Coupled_Immunity' && isRequiredParameter(col.key)) {
					input.required = true;
				  }

				  cell.appendChild(input);
				}
			  });

			  const actionCell = row.insertCell();
			  actionCell.style.whiteSpace = 'nowrap';
			  const copyBtn = document.createElement('button');
			  copyBtn.className = 'btn-copy';
			  copyBtn.textContent = 'Copy';
			  copyBtn.onclick = function() { copyRow(this, 'limit'); };
			  const deleteBtn = document.createElement('button');
			  deleteBtn.className = 'btn-remove';
			  deleteBtn.textContent = 'Delete';
			  deleteBtn.onclick = function() { removeRow(this); };
			  actionCell.appendChild(copyBtn);
			  actionCell.appendChild(deleteBtn);

			  setupParameterValidation(row);
			}

		function copyRow(btn, tableType) {
			const currentRow = btn.closest('tr');
			const inputs = currentRow.querySelectorAll('input[data-key], select[data-key]');
			const rowData = {};
			inputs.forEach(input => rowData[input.getAttribute('data-key')] = input.value);
			if (tableType === 'general') addGeneralRowDynamic(rowData);
			else if (tableType === 'limit') addLimitRowDynamic(rowData);
			const newRow = currentRow.parentNode.lastElementChild;
			newRow.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
			newRow.style.backgroundColor = '#E8F5E9';
			setTimeout(() => { newRow.style.backgroundColor = ''; }, 1000);
			showNotification('✅ Already Copy and add rows');
		}

		function removeRow(btn) {
			const tbody = btn.closest('tbody');
			if (tbody.rows.length <= 1) {
				showNotification('⚠️ Keep 1 row at least', 'error');
				return;
			}
			btn.closest('tr').remove();
			showNotification('✅ Already delete row');
		}

		function clearParameters() {
			if (!confirm('Confirm to Clear all parameter？')) return;
			
			// ✅ 清空全局数据
			parameterData.general = [];
			parameterData.limit = [];
			
			if (selectedLevel1 === 'Transient_Coupled_Immunity') {
				document.getElementById('limitTableBody').innerHTML = '';
				addLimitRowDynamic();
			} else {
				const generalTableBody = document.getElementById('generalTableBody');
				const limitTableBody = document.getElementById('limitTableBody');
				
				if (generalTableBody) {
					generalTableBody.innerHTML = '';
					addGeneralRowDynamic();
				}
				if (limitTableBody) {
					limitTableBody.innerHTML = '';
					addLimitRowDynamic();
				}
			}
			
			// ✅ 清空显示的汇总文本
			document.getElementById('cit_setting').value = '';
			
			showNotification('✅ 参数已清空');
		}

		function isRequiredParameter(paramName) {
			if (selectedLevel1 !== 'Transient_Coupled_Immunity' || !selectedLevel3) return false;
			const waveformConfig = config.waveformParameterMapping?.[selectedLevel1]?.[selectedLevel2]?.[selectedLevel3];
			if (!waveformConfig || !waveformConfig.requiredParams) return false;
			return waveformConfig.requiredParams.includes(paramName);
		}

		    // ========== 导出 General Parameters ==========
		    if (parameterData.general && parameterData.general.length > 0) {
		        const generalColumns = window.currentGeneralColumns || [];
		        const headerMapping = {};
		        generalColumns.forEach(col => headerMapping[col.key] = col.label);

		                newRow[headerMapping[key] || key] = value;
		            }
		            return newRow;
		        });
		        
		        const generalSheet = XLSX.utils.json_to_sheet(generalDataWithLabels);
		        
		        // ✅ 美化 SettingPar 表
		        beautifyParameterSheet(generalSheet, generalColumns);
		        
		        XLSX.utils.book_append_sheet(workbook, generalSheet, "SettingPar");
		    }
		    
		    // ========== 导出 Limit Parameters ==========
		    if (parameterData.limit && parameterData.limit.length > 0) {
		        const limitColumns = window.currentLimitColumns || [];
		        const headerMapping = {};
		        limitColumns.forEach(col => headerMapping[col.key] = col.label);
		        
		        const limitDataWithLabels = parameterData.limit.map(row => {
		            const newRow = {};
		            for (const [key, value] of Object.entries(row)) {
		                newRow[headerMapping[key] || key] = value;
		            }
		            return newRow;
		        });
		        
		        const limitSheet = XLSX.utils.json_to_sheet(limitDataWithLabels);
		        
		        // ✅ 美化 LimitLevel 表
		        beautifyParameterSheet(limitSheet, limitColumns);
		        
		        XLSX.utils.book_append_sheet(workbook, limitSheet, "LimitLevel");
		    }
		}

		function beautifyParameterSheet(ws, columns) {
		    const range = XLSX.utils.decode_range(ws['!ref']);
		    
		    // 设置列宽（根据列数自动调整）
		    ws['!cols'] = columns.map(() => ({ wch: 15 }));
		    
		    // 遍历所有单元格，添加样式
		    for (let R = range.s.r; R <= range.e.r; ++R) {
		        for (let C = range.s.c; C <= range.e.c; ++C) {
		            const cellAddress = XLSX.utils.encode_cell({ r: R, c: C });
		            if (!ws[cellAddress]) continue;
		            
		            const cell = ws[cellAddress];
		            
		            // 初始化单元格样式
		            if (!cell.s) cell.s = {};
		            
		            // 🎨 表头行样式（第一行）
		            if (R === 0) {
		                cell.s = {
		                    fill: { fgColor: { rgb: "4472C4" } },  // 淡紫色背景
		                    font: { bold: true, sz: 11, color: { rgb: "FFFFFF" } },
		                    alignment: { horizontal: "center", vertical: "center", wrapText: true },
		                    border: {
		                        top: { style: "medium", color: { rgb: "000000" } },
		                        bottom: { style: "medium", color: { rgb: "000000" } },
		                        left: { style: "thin", color: { rgb: "000000" } },
		                        right: { style: "thin", color: { rgb: "000000" } }
		                    }
		                };
		            }
		            // 🎨 数据行样式（奇数行）
		            else if (R % 2 === 1) {
		                cell.s = {
		                    fill: { fgColor: { rgb: "FFFFFF" } },  // 白色背景
		                    font: { sz: 10, color: { rgb: "000000" } },
		                    alignment: { horizontal: "center", vertical: "center" },
		                    border: {
		                        top: { style: "thin", color: { rgb: "DDDDDD" } },
		                        bottom: { style: "thin", color: { rgb: "DDDDDD" } },
		                        left: { style: "thin", color: { rgb: "DDDDDD" } },
		                        right: { style: "thin", color: { rgb: "DDDDDD" } }
		                    }
		                };
		            }
		            // 🎨 数据行样式（偶数行）- 斑马纹
		            else {
		                cell.s = {
		                    fill: { fgColor: { rgb: "F5F5F5" } },  // 浅灰色背景
		                    font: { sz: 10, color: { rgb: "000000" } },
		                    alignment: { horizontal: "center", vertical: "center" },
		                    border: {
		                        top: { style: "thin", color: { rgb: "DDDDDD" } },
		                        bottom: { style: "thin", color: { rgb: "DDDDDD" } },
		                        left: { style: "thin", color: { rgb: "DDDDDD" } },
		                        right: { style: "thin", color: { rgb: "DDDDDD" } }
		                    }
		                };
		            }
		        }
		    }
		    
		    // 设置行高
		    if (!ws['!rows']) ws['!rows'] = [];
		    ws['!rows'][0] = { hpt: 25 };  // 表头行高
		    for (let i = 1; i <= range.e.r; i++) {
		        ws['!rows'][i] = { hpt: 20 };  // 数据行行高
		    }
		    
		    // 冻结首行（表头）
		    ws['!freeze'] = { xSplit: 0, ySplit: 1 };
		}
